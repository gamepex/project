########## 직원 관리 테이블 ##########
create table tbl_staff ( -- tbl_staff 테이블 생성한다.
	staff_id    char(20)    comment '아이디',
	staff_pw    varchar(100)    not null    comment '비밀번호',
	staff_name    varchar(30)    not null    comment '이름',
	staff_gender    char(1)    comment '성별(m:남성, w:여성)',
	staff_part    varchar(10)    comment '부서',
	staff_position    varchar(10)    comment '직급',
	staff_mail    varchar(50)    not null    comment '메일',
	staff_phone    char(13)    not null    comment '전화번호',
	staff_approval    int    default 0    comment '승인여부(1:승인, 0:미승인)',
	staff_regdate   datetime    comment '등록일',
	staff_moddate   datetime    comment '수정일',
	primary key (staff_id)
) comment '직원 테이블';

########## 도서 관리 테이블 ##########
create table tbl_book ( -- tbl_book 테이블 생성한다.
	bk_no    int    auto_increment    comment '도서번호',
	staff_id    char(20)    not null    comment '직원아이디',
	bk_name    varchar(50)    not null    comment '도서이름',
	bk_author    varchar(50)    not null    comment '저자',
	bk_publisher    varchar(50)    not null    comment '출판사',
	bk_year    char(10)    not null    comment '출판연도',
	bk_rental_price	int	comment '대여가격',
	bk_type		varchar(20)	comment '도서유형',
	bk_thumbnail    varchar(100)    comment '도서표지',
	bk_content	text		comment '도서상세설명',
	bk_isbn    varchar(30)    not null    comment 'ISBN',
	bk_call_number    varchar(30)    not null    comment '청구기호',
	bk_rental_able    int    default 0    comment '전시여부(0:입고전, 1:전시, 2:대여중)',
	bk_regdate   datetime    comment '등록일',
	bk_moddate   datetime    comment '수정일',
	primary key (bk_no),
	foreign key (staff_id) references tbl_staff (staff_id)
) comment '도서 테이블';





########## 회원 관리 테이블 ##########
create table tbl_user ( -- tbl_user 테이블 생성한다.
	user_id    char(20)    comment '아이디',
	user_pw    varchar(100)    not null    comment '비밀번호',
	user_name    varchar(30)    not null    comment '이름',
	user_nickname    varchar(30)    not null    comment '별칭',
	user_gender    char(1)    comment '성별(m:남성, w:여성)',
	user_mail    varchar(50)    not null    comment '메일',
	user_phone    char(13)    not null    comment '전화번호',
	user_zipcode    char(7)    comment '우편번호',
	user_addr    varchar(200)    comment '주소',
	user_state    int    default 0    comment '회원여부(1:회원, 0:비회원)',
	user_regdate   datetime    comment '등록일',
	user_moddate   datetime    comment '수정일',
	primary key (user_id),
	unique (user_nickname)
) comment '회원테이블';


########## 주소 테이블 ##########

	area3	char(40)	comment '동/면',
	area4	char(30)	comment '번지'
