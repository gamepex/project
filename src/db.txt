create database gamepex;

CREATE TABLE tbl_staff (
 staff_no INT AUTO_INCREMENT COMMENT '사원번호',
 staff_id VARCHAR(20) NOT NULL COMMENT '아이디',
 staff_pw CHAR(60) NOT NULL COMMENT '비밀번호',
 staff_name VARCHAR(30) NOT NULL COMMENT '이름',
 staff_gender CHAR(1) COMMENT '성별',
 staff_birth DATE COMMENT '생년월일',
 staff_mail VARCHAR(50) NOT NULL COMMENT '이메일',
 staff_phone CHAR(13) NOT NULL COMMENT '연락처',
 staff_part VARCHAR(10) COMMENT '부서',
 staff_position VARCHAR(10) COMMENT '직급',
 staff_state INT NOT NULL DEFAULT 0 COMMENT '상태',
 staff_regdate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '등록일',
 staff_moddate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정일',
 PRIMARY KEY (staff_no),
 UNIQUE KEY uk_staff_id (staff_id),
 UNIQUE KEY uk_staff_mail (staff_mail),
 UNIQUE KEY uk_staff_phone (staff_phone)
) COMMENT '직원 테이블';

CREATE TABLE tbl_member (
 mb_id VARCHAR(20) NOT NULL COMMENT '아이디',
 mb_pw CHAR(60) NOT NULL COMMENT '비밀번호',
 mb_nickname VARCHAR(30) NOT NULL COMMENT '닉네임',
 mb_name VARCHAR(30) NOT NULL COMMENT '이름',
 mb_gender CHAR(1) COMMENT '성별',
 mb_birth DATE COMMENT '생년월일',
 mb_mail VARCHAR(50) NOT NULL COMMENT '이메일',
 mb_phone CHAR(13) NOT NULL COMMENT '연락처',
 mb_zipcode CHAR(5) NOT NULL COMMENT '우편번호',
 mb_addr VARCHAR(200) NOT NULL COMMENT '주소',
 mb_addr_detail VARCHAR(100) NOT NULL COMMENT '상세주소',
 mb_grade INT NOT NULL DEFAULT 0 COMMENT '등급',
 mb_state INT NOT NULL DEFAULT 0 COMMENT '상태',
 mb_regdate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '등록일',
 mb_moddate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정일',
 PRIMARY KEY (mb_id),
 UNIQUE KEY uk_member_nickname (mb_nickname),
 UNIQUE KEY uk_member_mail (mb_mail),
UNIQUE KEY uk_member_phone (mb_phone)
) COMMENT '회원 테이블';


CREATE TABLE `tbl_rental` (
  `rt_no` int NOT NULL AUTO_INCREMENT COMMENT '렌탈번호',
  `con_serial` varchar(30) NULL COMMENT '콘솔 일련번호',
  `ttl_serial` varchar(30) NULL COMMENT '타이틀 일련번호',
  `mb_id` varchar(20) NOT NULL COMMENT '회원 아이디',
  `rt_days` int NOT NULL COMMENT '대여일수',
  `rt_shipfee` int NOT NULL COMMENT '배송비',
  `rt_quantities` int DEFAULT NULL COMMENT '수량',
  `rt_amount` int NOT NULL COMMENT '결제총액',
  `rt_regdate` datetime DEFAULT NULL COMMENT '주문날짜',
  `rt_payment` int DEFAULT NULL COMMENT '결제방식 (0:카드 1:계좌이체 2:무통장입금 3:그 외)',
  `rt_invoice` varchar(20) DEFAULT NULL COMMENT '송장번호',
  `rt_state` int DEFAULT '0' COMMENT '주문상태 (0:pending 1: delivering 2:shipped 3:cancelled)',
  `rt_name` varchar(30) NOT NULL COMMENT '받는분성함',
  `rt_zipcode` char(5) NOT NULL COMMENT '우편번호',
  `rt_addr` varchar(200) NOT NULL COMMENT '받는분주소',
  `rt_addrdet` varchar(100) NOT NULL COMMENT '받는분 상세주소',
  `rt_tel` char(13) NOT NULL COMMENT '받는분전화번호',
  `rt_req` varchar(100) DEFAULT NULL COMMENT '배송요청사항',
  `rt_startdate` date NOT NULL COMMENT '시작일',
  `rt_enddate` date NOT NULL COMMENT '종료일',
  `rt_rtndate` date DEFAULT NULL COMMENT '반납일(관리자 권한)',
  `rt_moddate` datetime DEFAULT NULL COMMENT '수정일',
  PRIMARY KEY (`rt_no`),
  UNIQUE KEY `UQ_invoice` (`rt_invoice`),
  KEY `tbl_rental_tbl_member_FK` (`mb_id`),
  KEY `tbl_rental_tbl_console_FK` (`con_serial`),
  KEY `tbl_rental_tbl_title_FK` (`ttl_serial`),
  CONSTRAINT `tbl_rental_tbl_console_FK` FOREIGN KEY (`con_serial`) REFERENCES `tbl_console` (`con_serial`),
  CONSTRAINT `tbl_rental_tbl_member_FK` FOREIGN KEY (`mb_id`) REFERENCES `tbl_member` (`mb_id`),
  CONSTRAINT `tbl_rental_tbl_title_FK` FOREIGN KEY (`ttl_serial`) REFERENCES `tbl_title` (`ttl_serial`),
  CONSTRAINT `tbl_rental_chk_1` CHECK ((`rt_days` >= 3))
) ENGINE=InnoDB AUTO_INCREMENT=21 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='렌탈 테이블';
